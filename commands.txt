# Run from interceptV2DualDis.

# --- Numeric check: mean/peak |joint_vel| on the 8 arm columns (0-7) ---
# BEFORE (true originals, motions/data_pre_armdamp_backup/):
uv run python -c "
import numpy as np, glob
ARM_COLS = list(range(8))  # Left/Right Shoulder_Pitch/Roll, Elbow_Pitch/Yaw
for f in sorted(glob.glob('src/simple_goalkeeper/motions/data_pre_armdamp_backup/*.npz')):
    d = np.load(f)
    v = np.abs(d['joint_vel'][:, ARM_COLS])
    print(f'{f}: mean={v.mean():.3f} rad/s  peak={v.max():.3f} rad/s')
"

# AFTER (dampened, live in motions/data/):
uv run python -c "
import numpy as np, glob
ARM_COLS = list(range(8))
for f in sorted(glob.glob('src/simple_goalkeeper/motions/data/new_doublestep*.npz')):
    d = np.load(f)
    v = np.abs(d['joint_vel'][:, ARM_COLS])
    print(f'{f}: mean={v.mean():.3f} rad/s  peak={v.max():.3f} rad/s')
"
# Measured result: mean arm speed down 31-55% (0.41-0.69 -> 0.29-0.31 rad/s),
